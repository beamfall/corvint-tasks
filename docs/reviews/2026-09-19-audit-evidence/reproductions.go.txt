func TestAuditProjectionEditNotAdopted(t *testing.T) {
 repo, _ := initialized(t)
 created := mutate(t, repo, envelope("audit-create", mutation.OpCreate, "", "", createPayload("original")))
 loaded, err := intent.Load(repo.PrimaryWorktree); if err != nil { t.Fatal(err) }
 rec, _ := loaded.Inventory.Get(created.Ticket)
 rec.Title = "manually injected title"
 fixture.Write(t, filepath.Join(repo.PrimaryWorktree, intent.Dir, intent.TicketsDir, rec.TicketID.Local+".json"), rec.Encode())
 report, err := store.Mutate(context.Background(), repo, operator(), envelope("audit-prioritize", mutation.OpPrioritize, created.Ticket, "1", obj("priority",str("P1"),"order",str("0"))), now(t))
 if err == nil && report.Outcome.Outcome == mutation.OutcomeCompleted { t.Fatalf("BUG: divergent title silently adopted in receipt=%s resulting revision=%s", report.Receipt, *report.Outcome.ResultingRevision) }
}

func TestAuditInvalidInitDoesNotStrandStore(t *testing.T) {
 repo:=fixture.TempRepo(t)
 fixture.Write(t,filepath.Join(repo.IntentDir,"queue.json"),fixture.QueueBytes())
 fixture.Write(t,filepath.Join(repo.IntentDir,"policy.json"),fixture.PolicyBytes())
 resolved,err:=intent.Resolve(repo.Root); if err!=nil {t.Fatal(err)}
 bad,err:=store.Init(context.Background(),resolved,mutation.Binding{ID:"tester",Role:"INVALID"},"audit-invalid",now(t)); if err!=nil {t.Fatal(err)}
 good,err:=store.Init(context.Background(),resolved,operator(),"audit-valid",now(t)); if err!=nil {t.Fatal(err)}
 if good.Receipt=="" {t.Fatalf("BUG: invalid init=%s then valid init=%s receipt=%q detail=%q",bad.Outcome.Outcome,good.Outcome.Outcome,good.Receipt,good.Detail)}
}
